1//! Async Rust framework for the [Telegram Bot API](https://core.telegram.org/bots/api).
2//!
3//! rustigram re-exports all sub-crates through a single dependency and a
4//! convenient [`prelude`] module covering the most common types.
5//!
6//! # Getting started
7//!
8//! Add to `Cargo.toml`:
9//!
10//! ```toml
11//! [dependencies]
12//! rustigram = "0.12.0"
13//! tokio     = { version = "1", features = ["full"] }
14//! ```
15//!
16//! ```rust, no_run
17//! use rustigram::prelude::*;
18//!
19//! #[tokio::main]
20//! async fn main() -> anyhow::Result<()> {
21//!     let bot = Bot::new(std::env::var("BOT_TOKEN")?)?;
22//!
23//!     bot.dispatcher()
24//!         .on(filters::command("start"), handler_fn(start))
25//!         .on(filters::message(),        handler_fn(echo))
26//!         .build()
27//!         .polling()
28//!         .await?;
29//!
30//!     Ok(())
31//! }
32//!
33//! async fn start(ctx: Context) -> BotResult<()> {
34//!     if let Some(r) = ctx.reply("Hello!") { r.await?; }
35//!     Ok(())
36//! }
37//!
38//! async fn echo(ctx: Context) -> BotResult<()> {
39//!     if let (Some(text), Some(chat_id)) = (ctx.text(), ctx.chat_id()) {
40//!         ctx.bot.send_message(chat_id, text).await?;
41//!     }
42//!     Ok(())
43//! }
44//! ```
45//!
46//! # Crate structure
47//!
48//! | Crate | Purpose |
49//! |---|---|
50//! | [`api`] | HTTP client and typed method builders |
51//! | [`bot`] | Dispatcher, filters, handlers, FSM |
52//! | [`types`] | All Bot API type definitions |
53//!
54#![doc(html_root_url = "https://docs.rs/rustigram/0.12.0")]
55
56pub use rustigram_api as api;
57pub use rustigram_bot as bot;
58pub use rustigram_types as types;

69    pub use rustigram_types::{

73    };
