1//! Error types for `rustigram-miniapp`.
2//!
3//! All fallible operations in this crate return [`Result<T>`], which is an
4//! alias for `std::result::Result<T, MiniAppError>`. Every variant maps to an
5//! appropriate HTTP status code via [`axum::response::IntoResponse`] so
6//! handlers can return errors directly without manual conversion.
7
8use axum::{
9    http::StatusCode,
10    response::{IntoResponse, Response},
11};
12use thiserror::Error;
13
14/// All error conditions produced by `rustigram-miniapp`.
15///
16/// Implements [`IntoResponse`] so variants can be returned directly from Axum
17/// handlers. Auth failures → `401 Unauthorized`; bad input → `400 Bad
18/// Request`; missing layer → `500 Internal Server Error`.
19#[derive(Debug, Error)]
20pub enum MiniAppError {
21    /// The initData query string was unparseable, or a required field
22    /// (`auth_date`, `hash`) was absent, or a sub-object field (`user`,
23    /// `chat`, `receiver`) contained invalid JSON.
24    #[error("malformed initData: {0}")]
25    MalformedInitData(String),
26
27    /// The HMAC-SHA256 hash did not match the expected value. The initData was
28    /// either tampered with or validated against the wrong bot token.
29    #[error("invalid HMAC hash")]
30    InvalidHmac,
31
32    /// The Ed25519 signature did not verify against Telegram's published
33    /// public key for the selected environment.
34    #[error("invalid Ed25519 signature")]
35    InvalidSignature,
36
37    /// The `auth_date` field is older than the configured `max_age_secs`.
38    /// The initData is considered stale and should be rejected.
39    #[error("initData expired")]
40    Expired,
41
42    /// No [`crate::extract::BotToken`] extension was found on the request.
43    /// The router is missing a [`crate::extract::BotTokenLayer`].
44    #[error("missing BotToken extension — add BotTokenLayer to the router")]
45    MissingBotToken,
46}
47
48/// Convenience alias used throughout this crate.
49pub type Result<T> = std::result::Result<T, MiniAppError>;
50
51impl IntoResponse for MiniAppError {
52    fn into_response(self) -> Response {
53        let status = match self {
54            Self::InvalidHmac | Self::InvalidSignature | Self::Expired => StatusCode::UNAUTHORIZED,
55            Self::MalformedInitData(_) => StatusCode::BAD_REQUEST,
56            Self::MissingBotToken => StatusCode::INTERNAL_SERVER_ERROR,
57        };
58        (status, self.to_string()).into_response()
59    }
60}
