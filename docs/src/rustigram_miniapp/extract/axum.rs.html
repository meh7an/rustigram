14/// Axum extractor that validates `initData` from the incoming request and
15/// provides a typed [`WebAppInitData`] to the handler.
16///
17/// Reads the raw initData string from — in priority order:
18/// 1. `X-Tma-Init-Data` header
19/// 2. `Authorization: tma <data>` header
20///
21/// Validation method is auto-detected: if the raw string contains a
22/// `signature` field, Ed25519 is used; otherwise HMAC-SHA256.
23///
24/// Requires [`BotTokenLayer`] on the router. Returns `401` on auth failure,
25/// `400` on malformed input, `500` if the layer is missing.
26///
27/// # Example
28///
29/// ```rust,ignore
30/// async fn handler(TmaInitData(data): TmaInitData) {
31///     println!("user: {:?}", data.user);
32/// }
33/// ```
34pub struct TmaInitData(pub WebAppInitData);
35
36impl<S> FromRequestParts<S> for TmaInitData
37where
38    S: Send + Sync,
39{
40    type Rejection = MiniAppError;
41
42    async fn from_request_parts(parts: &mut Parts, _state: &S) -> Result<Self> {
43        let raw = read_init_data_header(parts)?;
44
45        let token = parts
46            .extensions
47            .get::<BotToken>()
48            .ok_or(MiniAppError::MissingBotToken)?;
49
50        // Auto-detect validation method: Ed25519 when `signature` is present.
51        let params = parse_raw(&raw);
52        let has_signature = params.iter().any(|(k, _)| k == "signature");
53
54        let init_data = if has_signature {
55            let bot_id = extract_bot_id(&token.0)?;
56            validate_ed25519(&raw, bot_id, Ed25519ValidateOpts::default())?
57        } else {
58            validate_hmac(&raw, &token.0, HmacValidateOpts::default())?
59        };
60
61        Ok(TmaInitData(init_data))
62    }
63}
64
65/// Reads the raw initData string, or reports which headers were expected.
66fn read_init_data_header(parts: &Parts) -> Result<String> {
67    super::init_data_from_headers(&parts.headers).ok_or_else(|| {
68        MiniAppError::MalformedInitData(
69            "missing X-Tma-Init-Data header (or Authorization: tma <data>)".into(),
70        )
71    })
72}
73
74/// Extracts the numeric bot ID from the bot token (`123456789:ABC...` → `123456789`).
75fn extract_bot_id(token: &str) -> Result<i64> {
76    token
77        .split(':')
78        .next()
79        .and_then(|s| s.parse().ok())
80        .ok_or_else(|| MiniAppError::MalformedInitData("cannot extract bot ID from token".into()))
81}

107    fn app_with_layer() -> Router {

113    fn build_valid_init_data(token: &str) -> String {
114        let auth_date = std::time::SystemTime::now()
115            .duration_since(std::time::UNIX_EPOCH)
116            .unwrap()
117            .as_secs()
118            .to_string();
119        let user_json = r#"{"id":42,"first_name":"Mehran"}"#;
120
121        let mut entries = [
122            ("auth_date", auth_date.clone()),
123            ("query_id", "test_query_001".to_string()),
124            ("user", user_json.to_string()),
125        ];
126        entries.sort_by_key(|(a, _)| *a);
127        let dcs = entries
128            .iter()
129            .map(|(k, v)| format!("{k}={v}"))
130            .collect::<Vec<_>>()
131            .join("\n");
132
133        let mut mac = HmacSha256::new_from_slice(b"WebAppData").unwrap();
134        mac.update(token.as_bytes());
135        let sk = mac.finalize().into_bytes();
136        let mut mac = HmacSha256::new_from_slice(&sk).unwrap();
137        mac.update(dcs.as_bytes());
138        let hash = hex::encode(mac.finalize().into_bytes());
139
140        form_urlencoded::Serializer::new(String::new())
141            .append_pair("auth_date", &auth_date)
142            .append_pair("query_id", "test_query_001")
143            .append_pair("user", user_json)
144            .append_pair("hash", &hash)
145            .finish()
146    }

149    async fn valid_header_returns_200() {
150        let raw = build_valid_init_data(BOT_TOKEN);
151        let req = Request::builder()
