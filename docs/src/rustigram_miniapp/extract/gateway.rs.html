1//! Tower middleware that signs validated requests for downstream trust.
2//!
3//! In Mode 3 (Rust gateway + SolidStart BFF), Rust validates initData via
4//! [`super::TmaInitData`] and then needs to prove to SolidStart that the
5//! request was validated — without sharing the bot token.
6//!
7//! [`TmaGatewayLayer`] adds an `X-Tma-Gateway` header containing an
8//! HMAC-SHA256 of the raw initData string signed with [`GatewaySecret`].
9//! SolidStart verifies this signature via `createTmaMiddleware(null, {
10//! gatewaySecret })` and only parses requests that carry a valid header.
11//!
12//! # Security model
13//!
14//! - `BOT_TOKEN` — stays in Rust only, never reaches SolidStart.
15//! - `GATEWAY_SECRET` — shared between Rust and SolidStart. Proves a
16//!   request passed through the validated Rust gateway.
17//! - Without `GATEWAY_SECRET`, trust is network-level only (VPC / firewall).
18//!   Add it for cryptographic proof when network isolation alone is not
19//!   sufficient.
20//!
21//! # Layer ordering
22//!
23//! Add [`TmaGatewayLayer`] **outside** (after) [`super::BotTokenLayer`] so
24//! the bot token is available when [`super::TmaInitData`] runs:
25//!
26//! ```rust,ignore
27//! Router::new()
28//!     .route("/tma", post(handler))
29//!     .layer(TmaGatewayLayer(GatewaySecret(
30//!         std::env::var("GATEWAY_SECRET").unwrap(),
31//!     )))
32//!     .layer(BotTokenLayer(BotToken(
33//!         std::env::var("BOT_TOKEN").unwrap(),
34//!     )));
35//! ```
36
37use axum::http::{HeaderName, HeaderValue, Request};
38use hmac::{Hmac, Mac};
39use sha2::Sha256;
40use std::task::{Context, Poll};
41use tower::{Layer, Service};
42
43static GATEWAY_HEADER: HeaderName = HeaderName::from_static("x-tma-gateway");
44
45/// Tower layer that adds an HMAC-signed `X-Tma-Gateway` header to every
46/// request that has passed [`TmaInitData`] extraction.
47///
48/// SolidStart (or any downstream) verifies this header via
49/// `createTmaMiddleware(null, { gatewaySecret })` so it can trust the
50/// request came from the validated Rust gateway rather than directly from
51/// the internet.
52///
53/// Add this layer AFTER `BotTokenLayer` on your router.
54#[derive(Clone)]
55pub struct TmaGatewayLayer(pub GatewaySecret);
56
57/// Newtype wrapping the gateway shared secret.
58#[derive(Debug, Clone)]
59pub struct GatewaySecret(pub String);
60
61impl<S> Layer<S> for TmaGatewayLayer {
62    type Service = TmaGatewayService<S>;
63
64    fn layer(&self, inner: S) -> Self::Service {
65        TmaGatewayService {
66            inner,
67            secret: self.0.clone(),
68        }
69    }
70}
71
72/// Produced by [`TmaGatewayLayer`].
73///
74/// For each request carrying initData — in the same headers
75/// [`super::TmaInitData`] reads, `X-Tma-Init-Data` or `Authorization: tma` —
76/// computes
77/// `HMAC-SHA256(gateway_secret, raw_init_data)` and injects the result as
78/// `X-Tma-Gateway` before delegating to the inner service.
79///
80/// Requests without the initData header pass through unmodified — the
81/// downstream [`super::TmaInitData`] extractor will reject them if the
82/// route requires validation.
83#[derive(Clone)]
84pub struct TmaGatewayService<S> {
85    inner: S,
86    secret: GatewaySecret,
87}
88
89impl<S, ReqBody> Service<Request<ReqBody>> for TmaGatewayService<S>
90where
91    S: Service<Request<ReqBody>>,
92{
93    type Response = S::Response;
94    type Error = S::Error;
95    type Future = S::Future;
96
97    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
98        self.inner.poll_ready(cx)
99    }
100
101    fn call(&mut self, mut req: Request<ReqBody>) -> Self::Future {
102        // Sign the raw initData with the gateway secret so downstream can
103        // verify it without knowing the bot token.
104        if let Some(init_data) = super::init_data_from_headers(req.headers()) {
105            type HmacSha256 = Hmac<Sha256>;
106            let mut mac = HmacSha256::new_from_slice(self.secret.0.as_bytes())
107                .expect("HMAC accepts a key of any length");
108            mac.update(init_data.as_bytes());
109            let sig = hex::encode(mac.finalize().into_bytes());
110            if let Ok(val) = HeaderValue::from_str(&sig) {
111                req.headers_mut().insert(GATEWAY_HEADER.clone(), val);
112            }
113        }
114        self.inner.call(req)
115    }
116}

118#[cfg(test)]
119mod tests {
120    use super::*;
121    use axum::body::Body;
122    use axum::http::Request as HttpRequest;
123    use axum::routing::get;
124    use axum::Router;
125    use tower::ServiceExt;
126
127    const SECRET: &str = "gateway-secret";
128    const INIT_DATA: &str = "auth_date=1700000000&query_id=q&hash=abc";
129
130    /// Echoes back whatever `X-Tma-Gateway` the layer produced, so a test can
131    /// assert on it.
132    async fn echo(headers: axum::http::HeaderMap) -> String {
133        headers
134            .get("x-tma-gateway")
135            .and_then(|v| v.to_str().ok())
136            .unwrap_or("<unsigned>")
137            .to_owned()
138    }
139
140    fn app() -> Router {
141        Router::new()
142            .route("/", get(echo))
143            .layer(TmaGatewayLayer(GatewaySecret(SECRET.to_owned())))
144    }
145
146    fn expected_signature() -> String {
147        type HmacSha256 = Hmac<Sha256>;
148        let mut mac = HmacSha256::new_from_slice(SECRET.as_bytes()).unwrap();
149        mac.update(INIT_DATA.as_bytes());
150        hex::encode(mac.finalize().into_bytes())
151    }
152
153    async fn body_of(req: HttpRequest<Body>) -> String {
154        let resp = app().oneshot(req).await.unwrap();
155        let bytes = axum::body::to_bytes(resp.into_body(), usize::MAX)
156            .await
157            .unwrap();
158        String::from_utf8(bytes.to_vec()).unwrap()
159    }
160
161    /// The regression: the layer used to read `X-Telegram-Init-Data`, which
162    /// nothing in the crate sets, so it never signed anything.
163    #[tokio::test]
164    async fn signs_the_header_the_extractor_reads() {
165        let req = HttpRequest::builder()
166            .uri("/")
167            .header("x-tma-init-data", INIT_DATA)
168            .body(Body::empty())
169            .unwrap();
170        assert_eq!(body_of(req).await, expected_signature());
171    }
172
173    /// The `Authorization: tma` fallback must sign identically — the downstream
174    /// verifies against the raw initData, not against which header carried it.
175    #[tokio::test]
176    async fn authorization_form_signs_identically() {
177        let req = HttpRequest::builder()
178            .uri("/")
179            .header("authorization", format!("tma {INIT_DATA}"))
180            .body(Body::empty())
181            .unwrap();
182        assert_eq!(body_of(req).await, expected_signature());
183    }
184
185    /// Documented behaviour: requests without initData pass through unsigned,
186    /// and the downstream extractor is what rejects them.
187    #[tokio::test]
188    async fn requests_without_init_data_pass_through_unsigned() {
189        let req = HttpRequest::builder().uri("/").body(Body::empty()).unwrap();
190        assert_eq!(body_of(req).await, "<unsigned>");
191    }
192
193    /// The old header must not be honoured, or the bug silently survives.
194    #[tokio::test]
195    async fn the_old_wrong_header_is_not_signed() {
196        let req = HttpRequest::builder()
197            .uri("/")
198            .header("x-telegram-init-data", INIT_DATA)
199            .body(Body::empty())
200            .unwrap();
201        assert_eq!(body_of(req).await, "<unsigned>");
202    }
