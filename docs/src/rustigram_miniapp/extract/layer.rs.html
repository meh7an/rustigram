1//! [`BotTokenLayer`] Tower middleware implementation.
2
3use std::task::{Context, Poll};
4
5use axum::http::Request;
6use tower::{Layer, Service};
7
8use super::BotToken;
9
10/// Tower middleware layer that injects a [`BotToken`] extension into every
11/// request.
12///
13/// Add this to your Axum router so that [`TmaInitData`] can read the token
14/// during extraction without coupling handlers to a specific `AppState` shape.
15///
16/// # Example
17///
18/// ```rust,ignore
19/// let app = Router::new()
20///     .route("/tma", post(my_handler))
21///     .layer(BotTokenLayer(BotToken(std::env::var("BOT_TOKEN").unwrap())));
22/// ```
23#[derive(Clone)]
24pub struct BotTokenLayer(pub BotToken);
25
26impl<S> Layer<S> for BotTokenLayer {
27    type Service = BotTokenService<S>;
28
29    fn layer(&self, inner: S) -> Self::Service {
30        BotTokenService {
31            inner,
32            token: self.0.clone(),
33        }
34    }
35}
36
37/// Produced by [`BotTokenLayer`]. Clones the [`BotToken`] into each request's
38/// extension map before delegating to the inner service.
39#[derive(Clone)]
40pub struct BotTokenService<S> {
41    inner: S,
42    token: BotToken,
43}
44
45impl<S, ReqBody> Service<Request<ReqBody>> for BotTokenService<S>
46where
47    S: Service<Request<ReqBody>>,
48{
49    type Response = S::Response;
50    type Error = S::Error;
51    type Future = S::Future;
52
53    fn poll_ready(&mut self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
54        self.inner.poll_ready(cx)
55    }
56
57    fn call(&mut self, mut req: Request<ReqBody>) -> Self::Future {
58        req.extensions_mut().insert(self.token.clone());
59        self.inner.call(req)
60    }
61}
