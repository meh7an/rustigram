1//! Telegram Mini App initData types.
2//!
3//! Typed representations of the objects Telegram injects into
4//! `window.Telegram.WebApp.initData` when a Mini App is opened. All types in
5//! this module mirror the shapes defined in `@rustigram/tma-core`'s Zod
6//! schemas and are validated server-side by [`crate::validate_hmac`] or
7//! [`crate::validate_ed25519`] before being handed to application code.
8//!
9//! @see <https://core.telegram.org/bots/webapps#webappinitdata>
10
11use serde::{Deserialize, Serialize};

13/// Type of group chat represented by a [`WebAppChat`].
14///
15/// Private chats and the "sender" context do not appear here — see
16/// [`InitDataChatType`] for the full launch-context enum.
17///
18/// @see <https://core.telegram.org/bots/webapps#webappchat>
19#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
20#[serde(rename_all = "snake_case")]
21#[cfg_attr(feature = "ts", derive(ts_rs::TS))]
22#[cfg_attr(feature = "ts", ts(export))]
23pub enum WebAppChatType {
24    /// Group chat.
25    Group,
26    /// Supergroup chat.
27    Supergroup,
28    /// Broadcast channel.
29    Channel,
30}

32/// Type of chat from which the Mini App was opened, as reported in
33/// [`WebAppInitData::chat_type`].
34///
35/// Wider than [`WebAppChatType`] — includes `sender` and `private` contexts
36/// that are not valid for the `chat` field itself.
37///
38/// @see <https://core.telegram.org/bots/webapps#webappinitdata>
39#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
40#[serde(rename_all = "snake_case")]
41#[cfg_attr(feature = "ts", derive(ts_rs::TS))]
42#[cfg_attr(feature = "ts", ts(export))]
43pub enum InitDataChatType {
44    /// Direct message from the attachment menu entry point.
45    Sender,
46    /// Private chat.
47    Private,
48    /// Group chat.
49    Group,
50    /// Supergroup chat.
51    Supergroup,
52    /// Broadcast channel.
53    Channel,
54}

56/// Contains data about the Mini App user.
57///
58/// Returned as the `user` or `receiver` field of [`WebAppInitData`].
59///
60/// # Flag fields
61///
62/// `is_premium`, `added_to_attachment_menu`, and `allows_write_to_pm` are
63/// `Option<bool>` but Telegram only ever sends them when `true`. An absent
64/// field is semantically `false`. `Some(false)` will never arrive from the
65/// platform.
66///
67/// @see <https://core.telegram.org/bots/webapps#webappuser>
68#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
69#[cfg_attr(feature = "ts", derive(ts_rs::TS))]
70#[cfg_attr(feature = "ts", ts(export))]
71pub struct WebAppUser {
72    /// A unique identifier for the user or bot. Has at most 52 significant
73    /// bits, so a 64-bit integer type is safe for storing this value.
74    #[cfg_attr(feature = "ts", ts(type = "number"))]
75    pub id: i64,
76
77    /// `true` if this user is a bot. Returned only in the `receiver` field —
78    /// never present on the initiating user.
79    #[serde(skip_serializing_if = "Option::is_none")]
80    #[cfg_attr(feature = "ts", ts(optional))]
81    pub is_bot: Option<bool>,
82
83    /// First name of the user or bot.
84    pub first_name: String,
85
86    /// Last name of the user or bot.
87    #[serde(skip_serializing_if = "Option::is_none")]
88    #[cfg_attr(feature = "ts", ts(optional))]
89    pub last_name: Option<String>,
90
91    /// Username of the user or bot.
92    #[serde(skip_serializing_if = "Option::is_none")]
93    #[cfg_attr(feature = "ts", ts(optional))]
94    pub username: Option<String>,
95
96    /// IETF language tag of the user's language. Returned only in the `user`
97    /// field, not in `receiver`.
98    #[serde(skip_serializing_if = "Option::is_none")]
99    #[cfg_attr(feature = "ts", ts(optional))]
100    pub language_code: Option<String>,
101
102    /// `true` if this user is a Telegram Premium subscriber. Omitted entirely
103    /// when `false` — Telegram never sends the field with a `false` value.
104    #[serde(skip_serializing_if = "Option::is_none")]
105    #[cfg_attr(feature = "ts", ts(optional))]
106    pub is_premium: Option<bool>,
107
108    /// `true` if this user has added the bot to the attachment menu. Omitted
109    /// entirely when `false`.
110    ///
111    /// @since Bot API 6.3
112    #[serde(skip_serializing_if = "Option::is_none")]
113    #[cfg_attr(feature = "ts", ts(optional))]
114    pub added_to_attachment_menu: Option<bool>,
115
116    /// `true` if this user has allowed the bot to message them. Omitted
117    /// entirely when `false`.
118    ///
119    /// @since Bot API 6.3
120    #[serde(skip_serializing_if = "Option::is_none")]
121    #[cfg_attr(feature = "ts", ts(optional))]
122    pub allows_write_to_pm: Option<bool>,
123
124    /// URL of the user's profile photo. Available to all Mini Apps when the
125    /// user's privacy settings permit it.
126    ///
127    /// @since Bot API 7.5
128    #[serde(skip_serializing_if = "Option::is_none")]
129    #[cfg_attr(feature = "ts", ts(optional))]
130    pub photo_url: Option<String>,
131}

133/// Represents a chat in which the Mini App was launched via the attachment menu.
134///
135/// @see <https://core.telegram.org/bots/webapps#webappchat>
136#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
137#[cfg_attr(feature = "ts", derive(ts_rs::TS))]
138#[cfg_attr(feature = "ts", ts(export))]
139pub struct WebAppChat {
140    /// Unique identifier for the chat. Has at most 52 significant bits, so a
141    /// 64-bit integer type is safe for storing this value.
142    #[cfg_attr(feature = "ts", ts(type = "number"))]
143    pub id: i64,
144
145    /// Type of the chat.
146    // Serialised as "type" to match the Telegram wire format.
147    #[serde(rename = "type")]
148    pub kind: WebAppChatType,
149
150    /// Title of the chat.
151    pub title: String,
152
153    /// Username of the chat.
154    #[serde(skip_serializing_if = "Option::is_none")]
155    #[cfg_attr(feature = "ts", ts(optional))]
156    pub username: Option<String>,
157
158    /// URL of the chat's photo.
159    #[serde(skip_serializing_if = "Option::is_none")]
160    #[cfg_attr(feature = "ts", ts(optional))]
161    pub photo_url: Option<String>,
162}

164/// Contains data transferred to the Mini App when it is opened. Empty if
165/// launched from a keyboard button or from inline mode.
166///
167/// **Warning:** Data from this object must not be trusted on the client.
168/// Only use `initData` on the bot's server after it has been validated
169/// against the bot token (HMAC-SHA256) or via Ed25519 for third-party use.
170///
171/// This is the output of a successful [`crate::validate_hmac`] or
172/// [`crate::validate_ed25519`] call. Never construct it directly — the
173/// validation functions are the only trusted source.
174///
175/// # Security
176///
177/// `hash` and `signature` are retained for completeness only. After
178/// validation they carry no further security value and should not be
179/// forwarded to untrusted parties.
180///
181/// @see <https://core.telegram.org/bots/webapps#webappinitdata>
182#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
183#[cfg_attr(feature = "ts", derive(ts_rs::TS))]
184#[cfg_attr(feature = "ts", ts(export))]
185pub struct WebAppInitData {
186    /// A unique identifier for the Mini App session, required for sending
187    /// messages via the `answerWebAppQuery` Bot API method.
188    #[serde(skip_serializing_if = "Option::is_none")]
189    #[cfg_attr(feature = "ts", ts(optional))]
190    pub query_id: Option<String>,
191
192    /// An object containing data about the current user.
193    #[serde(skip_serializing_if = "Option::is_none")]
194    #[cfg_attr(feature = "ts", ts(optional))]
195    pub user: Option<WebAppUser>,
196
197    /// An object containing data about the chat partner of the current user
198    /// in the chat where the bot was launched via the attachment menu.
199    /// Returned only for private chats and only for Mini Apps launched via
200    /// the attachment menu.
201    #[serde(skip_serializing_if = "Option::is_none")]
202    #[cfg_attr(feature = "ts", ts(optional))]
203    pub receiver: Option<WebAppUser>,
204
205    /// An object containing data about the chat where the bot was launched
206    /// via the attachment menu. Returned for supergroups, channels, and group
207    /// chats — only for Mini Apps launched via the attachment menu.
208    #[serde(skip_serializing_if = "Option::is_none")]
209    #[cfg_attr(feature = "ts", ts(optional))]
210    pub chat: Option<WebAppChat>,
211
212    /// Type of the chat from which the Mini App was opened. Returned only for
213    /// direct link Mini Apps and groups, supergroups, and channels.
214    #[serde(skip_serializing_if = "Option::is_none")]
215    #[cfg_attr(feature = "ts", ts(optional))]
216    pub chat_type: Option<InitDataChatType>,
217
218    /// Global identifier indicating the chat from which the Mini App was
219    /// opened. Returned only for direct link Mini Apps and groups,
220    /// supergroups, and channels.
221    #[serde(skip_serializing_if = "Option::is_none")]
222    #[cfg_attr(feature = "ts", ts(optional))]
223    pub chat_instance: Option<String>,
224
225    /// The value of the `startattach` or `startapp` parameter passed in the
226    /// link used to launch the Mini App.
227    #[serde(skip_serializing_if = "Option::is_none")]
228    #[cfg_attr(feature = "ts", ts(optional))]
229    pub start_param: Option<String>,
230
231    /// Time in seconds, after which a message can be sent via the
232    /// `answerWebAppQuery` method.
233    #[serde(skip_serializing_if = "Option::is_none")]
234    #[cfg_attr(feature = "ts", ts(optional, type = "number"))]
235    pub can_send_after: Option<i64>,
236
237    /// Unix timestamp of when the form was opened. Used to prevent replay
238    /// attacks. Required field — always present in valid initData.
239    #[cfg_attr(feature = "ts", ts(type = "number"))]
240    pub auth_date: i64,
241
242    /// An HMAC-SHA-256 hash of the `data-check-string` used to validate the
243    /// data against the bot token.
244    pub hash: String,
245
246    /// An Ed25519 signature of the `data-check-string` for third-party
247    /// validation without requiring the bot token.
248    ///
249    /// @since Bot API 8.0
250    #[serde(skip_serializing_if = "Option::is_none")]
251    #[cfg_attr(feature = "ts", ts(optional))]
252    pub signature: Option<String>,
253}
