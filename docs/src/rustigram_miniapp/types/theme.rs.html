1//! Telegram Mini App theme types.
2//!
3//! Typed representations of the theme parameters and color scheme Telegram
4//! exposes through `window.Telegram.WebApp.themeParams` and
5//! `window.Telegram.WebApp.colorScheme`. All color values are `#RRGGBB` hex
6//! strings. Fields are optional because older clients omit tokens they do not
7//! support — never treat any field as guaranteed.
8//!
9//! @see <https://core.telegram.org/bots/webapps#themeparams>
10
11use serde::{Deserialize, Serialize};
12
13/// Telegram UI color scheme reported by the client.
14///
15/// @see <https://core.telegram.org/bots/webapps#colorscheme>
16#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
17#[serde(rename_all = "snake_case")]
18#[cfg_attr(feature = "ts", derive(ts_rs::TS))]
19#[cfg_attr(feature = "ts", ts(export))]
20pub enum ColorScheme {
21    /// Light theme.
22    Light,
23    /// Dark theme.
24    Dark,
25}
26
27/// Contains the user's current theme settings used in the Telegram app.
28/// Mini Apps can adjust the appearance of the interface to match the
29/// Telegram user's app in real time.
30///
31/// All fields are optional — older clients omit fields they do not support.
32/// Never treat any field as guaranteed to be present.
33///
34/// Color values are always in `#RRGGBB` hex format when present.
35///
36/// @see <https://core.telegram.org/bots/webapps#themeparams>
37#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize)]
38#[cfg_attr(feature = "ts", derive(ts_rs::TS))]
39#[cfg_attr(feature = "ts", ts(export))]
40pub struct ThemeParams {
41    /// Background color in the `#RRGGBB` format.
42    /// CSS: `var(--tg-theme-bg-color)`
43    #[serde(skip_serializing_if = "Option::is_none")]
44    #[cfg_attr(feature = "ts", ts(optional))]
45    pub bg_color: Option<String>,
46
47    /// Main text color in the `#RRGGBB` format.
48    /// CSS: `var(--tg-theme-text-color)`
49    #[serde(skip_serializing_if = "Option::is_none")]
50    #[cfg_attr(feature = "ts", ts(optional))]
51    pub text_color: Option<String>,
52
53    /// Hint text color in the `#RRGGBB` format.
54    /// CSS: `var(--tg-theme-hint-color)`
55    #[serde(skip_serializing_if = "Option::is_none")]
56    #[cfg_attr(feature = "ts", ts(optional))]
57    pub hint_color: Option<String>,
58
59    /// Link color in the `#RRGGBB` format.
60    /// CSS: `var(--tg-theme-link-color)`
61    #[serde(skip_serializing_if = "Option::is_none")]
62    #[cfg_attr(feature = "ts", ts(optional))]
63    pub link_color: Option<String>,
64
65    /// Button color in the `#RRGGBB` format.
66    /// CSS: `var(--tg-theme-button-color)`
67    #[serde(skip_serializing_if = "Option::is_none")]
68    #[cfg_attr(feature = "ts", ts(optional))]
69    pub button_color: Option<String>,
70
71    /// Button text color in the `#RRGGBB` format.
72    /// CSS: `var(--tg-theme-button-text-color)`
73    #[serde(skip_serializing_if = "Option::is_none")]
74    #[cfg_attr(feature = "ts", ts(optional))]
75    pub button_text_color: Option<String>,

77    /// Secondary background color in the `#RRGGBB` format.

80    /// @since Bot API 6.1
81    #[serde(skip_serializing_if = "Option::is_none")]

88    /// @since Bot API 7.0
89    #[serde(skip_serializing_if = "Option::is_none")]
90    #[cfg_attr(feature = "ts", ts(optional))]
91    pub header_bg_color: Option<String>,
92
93    /// Bottom bar background color in the `#RRGGBB` format. Also applied to

97    /// @since Bot API 7.10
98    #[serde(skip_serializing_if = "Option::is_none")]
99    #[cfg_attr(feature = "ts", ts(optional))]
100    pub bottom_bar_bg_color: Option<String>,
101
102    /// Accent text color in the `#RRGGBB` format.
103    /// CSS: `var(--tg-theme-accent-text-color)`

106    #[serde(skip_serializing_if = "Option::is_none")]

108    pub accent_text_color: Option<String>,
